year = 2026

month_start = date(year, month, 1)

last_day = calendar.monthrange(year, month)[1]
month_end = date(year, month, last_day)

periods = []

current_start = month_start

while current_start <= month_end:
    days_until_friday = 4 - current_start.weekday()

    if days_until_friday < 0:
        days_until_friday += 7

    current_end = current_start + timedelta(days=days_until_friday)

    if current_end > month_end:

r = [["日期", "時數", "差異"]]


weekly_total = 0

for start, end in periods:
    data = df[
        (df["加班歸屬日"] >= pd.Timestamp(start))
        & (df["加班歸屬日"] <= pd.Timestamp(end))
    ]

    total_hours = data["加班計算時數"].sum()

    if start == month_start:
        total_hours += cross_month_hours

        date_text = (
            f"{start.strftime('%m/%d')}~"
            f"{end.strftime('%m/%d')}"
            f" (含跨日加班時數) "
        )
    else:
        date_text = f"{start.strftime('%m/%d')}~" f"{end.strftime('%m/%d')}"

    weekly_total += total_hours

    r.append([date_text, total_hours, ""])

    print(f"{date_text}: " f"{total_hours:g} H")

    grand_total = weekly_total

        "course_classroom": course_classroom,
    }

    # 處理 12 位學員資料
    for i in range(12):

        # 有輸入員工號才查詢
        if i < len(current_students):
            emp_no = str(current_students[i]).strip()
            employee = employee_df[
                employee_df["員工號"].astype(str).str.strip() == emp_no
            ]

            # 找到員工
            if not employee.empty:
                full_name = employee.iloc[0]["中英文姓名"]

            # 找不到員工
            else:
                full_name = "No Found"

        # 超過輸入人數就留空白
        else:
            full_name = ""

        # 建立：
        # number1
        # number2
        # number3
        # ...
        context2[f"number{i+1}"] = full_name

        # 日期轉換函式
        def get_days(time_text):

            result = []

            # 單一天日期
            # 例如：
            # 2026/06/18
            if "~" not in time_text:

                d = datetime.strptime(time_text, "%Y/%m/%d")

                result.append(d.strftime("%d/%b").upper())

                return result

            # 日期區間
            # 例如：
            # 2026/06/18~2026/06/28
            start_str, end_str = time_text.split("~")

            start_date = datetime.strptime(start_str.strip(), "%Y/%m/%d")

            end_date = datetime.strptime(end_str.strip(), "%Y/%m/%d")

            current = start_date

            # 從開始日期跑到結束日期
            while current <= end_date:

                # weekday()
                # 0=星期一
                # 1=星期二
                # ...
                # 5=星期六
                # 6=星期日

                # 只保留平日
                if current.weekday() < 5:

                    result.append(current.strftime("%d/%b").upper())

                # 日期加一天
                current += timedelta(days=1)

            return result

        # 取得日期清單
        date_list = get_days(input_date)

        # 寫入 Date1 ~ Date10
        for i in range(10):

            if i < len(date_list):

                context2[f"Date{i+1}"] = date_list[i]

            else:

                context2[f"Date{i+1}"] = ""
